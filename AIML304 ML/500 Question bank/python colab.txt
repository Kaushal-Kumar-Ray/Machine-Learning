It uses NASA’s earthaccess library and assumes you have a registered Earthdata account.

import pandas as pd
import xarray as xr
import numpy as np
from datetime import datetime

# Login (as above)
import earthaccess
session = earthaccess.login()

# Define bounding box for Nepal
# Approx: lat 26.3–30.4°N, lon 80.0–88.2°E
bbox = (80.0, 26.3, 88.2, 30.4)  # (lon_min, lat_min, lon_max, lat_max)

# Date span
start = "2005-01-01"
end = "2024-12-31"  # 20 years

# Search for monthly IMERG Final Run (research-quality) precipitation
granules = earthaccess.search_data(
    short_name="GPM_3IMERGHH_06_precipitationCal",
    version="06",
    bbox=bbox,
    temporal=(start, end)
)

print(f"Found {len(granules)} monthly IMERG precipitation granules.")

# Download all granules
out_dir = "downloads"
earthaccess.download(granules, out_dir)

# Process each NetCDF file and compute area average rainfall
records = []
for granule in granules:
    fn = granule["title"]  # or path in out_dir
    ds = xr.open_dataset(f"{out_dir}/{fn}")
    # variable "precipitationCal"
    pr = ds["precipitationCal"]
    # average over spatial dims:
    mean_rain = float(pr.where(
        (ds.lat >= bbox[1]) & (ds.lat <= bbox[3]) &
        (ds.lon >= bbox[0]) & (ds.lon <= bbox[2])
    ).mean().values)
    # date from ds.time
    dt = pd.to_datetime(ds.time.values).strftime("%Y-%m")
    records.append((dt, mean_rain))
    ds.close()

# Build DataFrame
df = pd.DataFrame(records, columns=["YearMonth", "Rainfall_mm"])
df = df.sort_values("YearMonth")
df["Year"] = df["YearMonth"].str[:4]
df["Month"] = df["YearMonth"].str[5:7].astype(int)
df = df[["Year", "Month", "Rainfall_mm"]]

# Save as plain-text file
with open("nepal_monthly_rainfall.txt", "w") as f:
    f.write("Year Month Rainfall_mm\n")
    for _, row in df.iterrows():
        f.write(f"{int(row.Year)} {row.Month} {row.Rainfall_mm:.1f}\n")

print("Saved nepal_monthly_rainfall.txt with", len(df), "entries.")

# Define Jaipur area of interest
jaipur = ee.Geometry.Rectangle([75.6, 26.7, 76.0, 27.1])

# Load Sentinel-2 surface reflectance image collection
image = ee.ImageCollection('COPERNICUS/S2_SR') \
    .filterBounds(jaipur) \
    .filterDate('2023-01-01', '2023-12-31') \
    .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 10)) \
    .median() \
    .clip(jaipur)

# Select relevant bands
bands = ['B2', 'B3', 'B4', 'B8']  # Blue, Green, Red, NIR
image = image.select(bands)
📌 Step 4: Define Training Points (sample LULC classes)

# Define land cover training samples
# Format: ee.Feature(ee.Geometry.Point([lon, lat]), {'landcover': class_id})

# Example: 0=Water, 1=Urban, 2=Vegetation
points = ee.FeatureCollection([
    ee.Feature(ee.Geometry.Point([75.81, 26.91]), {'landcover': 0}),  # Water
    ee.Feature(ee.Geometry.Point([75.84, 26.89]), {'landcover': 1}),  # Urban
    ee.Feature(ee.Geometry.Point([75.78, 26.95]), {'landcover': 2}),  # Vegetation
    ee.Feature(ee.Geometry.Point([75.76, 26.97]), {'landcover': 2}),
    ee.Feature(ee.Geometry.Point([75.88, 26.90]), {'landcover': 1}),
    ee.Feature(ee.Geometry.Point([75.83, 26.93]), {'landcover': 0}),
])

# Sample the image at those points
training = image.sampleRegions(
    collection=points,
    properties=['landcover'],
    scale=10
)
📌 Step 5: Train Random Forest Classifier

classifier = ee.Classifier.smileRandomForest(numberOfTrees=20).train(
    features=training,
    classProperty='landcover',
    inputProperties=bands
)

# Classify the image
classified = image.classify(classifier)
